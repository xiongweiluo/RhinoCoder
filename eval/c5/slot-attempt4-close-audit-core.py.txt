EXPECTED_BEFORE_SHA='6b3cb059893852dc963284f6deade3577803e9b6cc59c617dea7f763b46889f9'
CODES=frozenset({'close_audit_prior_inventory_changed','close_audit_scope','close_audit_sequence','close_audit_signature','close_audit_last_tail','close_audit_failed_request_dispatched','close_audit_last_capture','close_audit_hub_population','close_audit_unknown_hub_request','close_audit_original_stop_unknown','close_audit_original_receipt_unknown','close_audit_original_bytes_changed','close_audit_unexpected_append','close_audit_key_removal_population','close_audit_dispatcher_detach_unknown','close_audit_duplicate_intent','close_audit_private_copy_changed','close_audit_component_population','close_audit_raw_model_prompt','close_audit_previous_result_changed'})
class Refusal(ValueError):
    def __init__(self,code):
        assert code in CODES;self.code=code;super().__init__(code)
def require(v,code):
    if not v:raise Refusal(code)
def digest(v):return hashlib.sha256(json.dumps(v,sort_keys=True,separators=(',',':'),ensure_ascii=False,allow_nan=False).encode()).hexdigest()
def signature(key,payload):return hmac.new(key,json.dumps(payload,sort_keys=True,separators=(',',':'),ensure_ascii=False,allow_nan=False).encode(),hashlib.sha256).hexdigest()

def verify_tail(requests,responses,secret,bootstrap,source_sha,freeze_sha):
    require(len(requests)==len(responses)==45,'close_audit_sequence')
    seen=set()
    for sent,reply in zip(requests,responses):
        e=sent['envelope'];p=e['payload']
        require(sent['kind'] in {'control','execute'} and p['owner_freeze_sha256']==freeze_sha and p['task_sha256']==bootstrap['task_sha256'],'close_audit_scope')
        require(p['request_id'] not in seen and hmac.compare_digest(e['signature'],signature(secret,p)),'close_audit_signature');seen.add(p['request_id'])
    failed,closed,stopped=responses[-3:]
    require(requests[-3]['kind']=='execute' and failed.get('status')=='failed_no_retry' and failed.get('error_type')=='NativeError' and failed.get('error')=='expired/oversized handoff TTL' and failed.get('seq')==43,'close_audit_last_tail')
    require(all(r.get('status')!='failed_no_retry' for r in responses[:-3]),'close_audit_last_tail')
    require(requests[-2]['kind']==requests[-1]['kind']=='control' and requests[-2]['envelope']['payload']['action']=='close' and requests[-1]['envelope']['payload']['action']=='stop','close_audit_last_tail')
    require(closed.get('status')=='closed' and closed['close_capture']['fixture_registry_absent'] is True and closed['key_absence']=={'key_removed':True,'actual_absence_checked':True},'close_audit_last_tail')
    require(stopped.get('status')=='stopped' and stopped.get('had_failure') is True and stopped.get('hook_removed_in_same_callback') is True and closed['source_sha256']==stopped['source_sha256']==bootstrap['source_sha256']==source_sha and closed['active_sha256']==stopped['active_sha256']==bootstrap['active_sha256'],'close_audit_last_tail')
    captures=[r for r in responses[:-3] if r.get('status')=='captured']
    require(captures and captures[-1]['native']==closed['close_capture']['fixture_before_close_native'] and captures[-1]['state']==requests[-3]['envelope']['payload']['expected'],'close_audit_last_capture')
    return {'last_slot_original_stopped_with_failure':True,'last_slot_registry_absence_verified':True,'last_slot_hook_remove_verified':True,'original_result_stop_none_not_rewritten':True,'failed_handoff_ttl_refusal_retained':True}

def verify_delta(before,after,removed_expected,next_seq):
    removed=set(before)-set(after);added=set(after)-set(before)
    require(removed==set(removed_expected),'close_audit_key_removal_population')
    require(all(before[n]==after[n] for n in set(before)&set(after)),'close_audit_original_bytes_changed')
    exact={('hub-'+kind+'-%04d.json'%next_seq) for kind in ('request','response','client-time','service-time','host-check')}
    exact|={'hub-stop.json','host-continuity-terminal-receipt.json','host-continuity-seal.json'}
    required=exact-{'hub-client-time-%04d.json'%next_seq}
    require(required<=added and all(n in exact or re.fullmatch(r'host-continuity-\d{4}\.json',n) for n in added),'close_audit_unexpected_append')
    return {'original_nonkey_files_unchanged':True,'removed_dispatch_keys':len(removed),'added_cleanup_evidence_files':len(added),'audit_key_custody_preserved':True}

def copy_verified(state,destination,inventory):
    before=inventory(state);destination.mkdir(mode=0o700)
    for name,v in before.items():
        target=destination/name;target.parent.mkdir(parents=True,mode=0o700,exist_ok=True)
        fd=os.open(target,os.O_WRONLY|os.O_CREAT|os.O_EXCL|os.O_NOFOLLOW,0o600)
        with (state/name).open('rb') as src,os.fdopen(fd,'wb') as out:shutil.copyfileobj(src,out,1048576)
        require(hashlib.sha256(target.read_bytes()).hexdigest()==v['sha256'],'close_audit_private_copy_changed')
    require(inventory(state)==before,'close_audit_private_copy_changed');return before
